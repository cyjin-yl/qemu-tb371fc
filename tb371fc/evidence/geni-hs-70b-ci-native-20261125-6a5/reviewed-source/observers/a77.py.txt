#!/usr/bin/env python3
"""Bounded read-only observation of complete A77 job logs; no guest execution."""
from pathlib import Path
import argparse
import hashlib
import json
import re
import traceback
ROOT = None
HEAD = '70b387fb064cdf2099e85dd89213a5ea7e2c398f'
MERGE = None
TREE = '37bf696bc687422eac2bd967abda7d80af8e02b9'
BASE = None
SOURCE_LABEL = '70b387f'
JOBS = {}
VARIANTS = ('default', 'observed-revidr', 'l2-128k', 'l2-256k', 'l3-512k', 'l3-1m', 'l3-2m', 'l3-4m', 'crypto-off')
PCI = {'reg-read', 'init', 'legacy-read-write', 'mcq-read-write', 'no-prdt', 'sync-cache', 'long-prdt', 'short-prdt', 'mcq-cq-wraparound', 'query-flag', 'query-attribute', 'query-desciptor', 'wb-init', 'wb-read-write', 'hid', 'legacy-crypto-rejected', 'mcq-crypto-rejected'}

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def require(condition, message):
    if not condition:
        raise ValueError(message)


def bind_observation(path, digest, role):
    """A new explicit full-file binding is mandatory; no old artifact fallback."""
    global ROOT, MERGE, BASE, JOBS, OBSERVATION_BINDING
    require(isinstance(digest, str) and re.fullmatch('[0-9a-f]{64}', digest), 'binding SHA256 required')
    raw = Path(path).read_bytes()
    require(hashlib.sha256(raw).hexdigest() == digest, 'binding complete SHA256 differs')
    def pairs(items):
        value = {}
        for key, item in items:
            require(key not in value, 'duplicate binding key')
            value[key] = item
        return value
    binding = json.loads(raw, object_pairs_hook=pairs,
                         parse_constant=lambda value: (_ for _ in ()).throw(ValueError('nonfinite binding')))
    require(set(binding) == {'schema', 'source_head', 'source_tree', 'pr_merge_head',
                            'pr_base_head', 'evidence_root', 'jobs'}, 'binding fields differ')
    require(binding['schema'] == 'tb371fc-geni-ci-raw-observer-binding-v1' and
            binding['source_head'] == HEAD and binding['source_tree'] == TREE, 'new source binding differs')
    ROOT = Path(binding['evidence_root'])
    require(ROOT.is_absolute() and ROOT.is_dir(), 'explicit existing evidence directory required')
    MERGE, BASE = binding['pr_merge_head'], binding['pr_base_head']
    expected = {'push': {'run_id': 45599002342, 'board_job_id': 139152769553, 'a77_job_id': 139152769048},
                'pr': {'run_id': 45599016290, 'board_job_id': 139152837234, 'a77_job_id': 139152837634}}
    require(binding['jobs'] == expected, 'exact new run/job binding differs')
    require(all(type(value) is int for row in binding['jobs'].values() for value in row.values()),
            'run/job identity types differ')
    JOBS = {key: (row['run_id'], row[role + '_job_id'], HEAD if key == 'push' else MERGE)
            for key, row in binding['jobs'].items()}
    OBSERVATION_BINDING = {'path': str(path), 'bytes': len(raw), 'sha256': digest}
    require_bindings()

def require_bindings():
    for name, value in [('HEAD', HEAD), ('MERGE', MERGE), ('TREE', TREE), ('BASE', BASE)]:
        require(type(value) is str and re.fullmatch('[0-9a-f]{40}', value) is not None,
                name + ' remains unbound or invalid')
    require(SOURCE_LABEL == HEAD[:7], 'Source filename label differs from bound head')
    require(HEAD != MERGE and HEAD != BASE, 'Distinct source, merge and base identities required')
    require(set(JOBS) == {'push', 'pr'}, 'Exactly push and PR job bindings required')
    for role, checkout in [('push', HEAD), ('pr', MERGE)]:
        run, job, actual_checkout = JOBS[role]
        require(type(run) is int and run > 0 and type(job) is int and job > 0,
                role + ' run/job remain unbound or invalid')
        require(actual_checkout == checkout, role + ' checkout binding differs')
    require(JOBS['push'][0] != JOBS['pr'][0] and JOBS['push'][1] != JOBS['pr'][1],
            'Distinct push and PR run/job bindings required')

def observe(run, job, event, checkout):
    metadata = json.loads((ROOT / f'a77-run-{run}-metadata.json').read_text())
    require(metadata['head_sha'] == HEAD and metadata['event'] == event, 'Run head/event identity mismatch')
    require(metadata['id'] == run and metadata['run_attempt'] == 1, 'Exact A77 run and attempt identity mismatch')
    require(metadata['status'] == 'completed' and metadata['conclusion'] == 'success', 'A77 workflow run is not terminal success')
    jobs = json.loads((ROOT / f'a77-{job}-jobs.json').read_text())['jobs']
    matched = [x for x in jobs if x['id'] == job]
    require(len(matched) == 1, 'Missing unique expected job')
    actual = matched[0]
    require(actual['run_id'] == run and actual['name'] == 'cortex-a77-architecture', 'Job run/name mismatch')
    require(actual['status'] == 'completed' and actual['conclusion'] == 'success', 'A77 job is not completed success')
    require(all((s['status'] == 'completed' and s['conclusion'] == 'success' for s in actual['steps'])), 'A77 job contains an incomplete/failed/skipped step')
    path = ROOT / f'a77-{job}.log'
    raw = path.read_bytes()
    text = raw.decode('utf-8')
    require('�' not in text, 'Decoded log has replacement characters')
    raw_lines = text.splitlines()
    lines = [re.sub('^\\ufeff?\\d{4}-\\d\\d-\\d\\dT\\d\\d:\\d\\d:\\d\\d\\.\\d+Z ', '', l) for l in raw_lines]
    checkout_indices = [i for i, l in enumerate(lines) if l == '[command]/usr/bin/git log -1 --format=%H']
    require(len(checkout_indices) == 1, 'No unique checkout SHA observation')
    require(lines[checkout_indices[0] + 1] == checkout, 'Actual checkout SHA differs')
    guest = []
    pci = []
    invalid = []
    verified = []
    for i, line in enumerate(lines):
        match = re.fullmatch('PASS ([a-z0-9-]+): ([0-9]+) guest checks', line)
        if match:
            guest.append({'name': match[1], 'guest_checks': int(match[2]), 'line_number': i + 1, 'raw_line': raw_lines[i]})
        match = re.fullmatch('PASS /aarch64/virt/generic-pcihost/pci-bus-generic/pci-bus/ufs/ufs-tests/([a-z0-9-]+)', line)
        if match:
            pci.append({'name': match[1], 'line_number': i + 1, 'raw_line': raw_lines[i]})
        if line.startswith('PASS rejected '):
            invalid.append({'options': line[len('PASS rejected '):], 'line_number': i + 1, 'raw_line': raw_lines[i]})
        if line == 'Verified 852 real guest checks and 2 rejected CPU configurations':
            verified.append({'line_number': i + 1, 'raw_line': raw_lines[i]})
    require(tuple((x['name'] for x in guest)) == VARIANTS, 'Nine exact configurations absent or duplicated')
    require(all((x['guest_checks'] >= 75 for x in guest)), 'Variant minimum check count failed')
    require(sum((x['guest_checks'] for x in guest)) == 852, 'Actual variant counts do not total 852')
    require(len(pci) == 17 and {x['name'] for x in pci} == PCI, 'Seventeen exact PCI cases absent or duplicated')
    require([x['options'] for x in invalid] == [',l2-cache-size=1M', ',l3-cache-size=128K'], 'Invalid configuration rejection evidence differs')
    require(len(verified) == 1, 'No unique executed final verification line')
    first, last = (pci[0]['line_number'] - 1, verified[0]['line_number'])
    forbidden = [raw_lines[i] for i in range(first, last) if re.match('^(FAIL\\b|SKIP\\b|not ok\\b|Bail out!|##\\[error\\])', lines[i])]
    require(not forbidden, 'Failure/skip marker in test output interval')
    return {'run_id': run, 'job_id': job, 'event': event, 'requested_source_head': HEAD, 'actual_checkout': checkout, 'job_completed_success': True, 'all_job_steps_completed_success': True, 'log_path': str(path), 'log_bytes': len(raw), 'log_sha256': sha(raw), 'log_representation': 'Complete UTF-8 job text as returned by GitHub connector; BOM and newlines retained', 'no_utf8_replacement_characters': True, 'checkout_raw_line': raw_lines[checkout_indices[0] + 1], 'variants': guest, 'guest_checks': 852, 'invalid_configurations': invalid, 'pci_ufs_cases': pci, 'shared_pci_ufs_case_count': 17, 'executed_final_verification': verified[0], 'runtime_failure_or_skip_markers': forbidden}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--binding', required=True)
    parser.add_argument('--binding-sha256', required=True)
    args = parser.parse_args()
    bind_observation(args.binding, args.binding_sha256, 'a77')
    out = ROOT / f'genihs-{SOURCE_LABEL}-a77-rawlog-audit-001.json'
    result = {'schema': 'tb371fc-geni-hs-a77-rawlog-observation-v1', 'passed': False, 'auditor_sha256': sha(Path(__file__).read_bytes()), 'source_head': HEAD, 'runs': [], 'scope': 'Bounded raw job-log evidence, not a re-execution or a full CPU artifact audit.'}
    try:
        require_bindings()
        head = json.loads((ROOT / f'git-commit-{HEAD}.json').read_text())
        merge = json.loads((ROOT / f'git-commit-{MERGE}.json').read_text())
        require(head['sha'] == HEAD and merge['sha'] == MERGE, 'Source and merge commit identities differ')
        require(head['tree']['sha'] == merge['tree']['sha'] == TREE, 'PR merge tree differs from bound qualified head')
        require([p['sha'] for p in merge['parents']] == [BASE, HEAD], 'PR merge parents differ')
        result['observation_binding'] = OBSERVATION_BINDING
        result['matching_push_pr_tree'] = head['tree']['sha']
        result['runs'].append(observe(JOBS['push'][0], JOBS['push'][1], 'push', JOBS['push'][2]))
        result['runs'].append(observe(JOBS['pr'][0], JOBS['pr'][1], 'pull_request', JOBS['pr'][2]))
        result['source_files'] = []
        for path, expected in [(ROOT / 'test_a77_cpu.py', '1969cdb5553abfac6e7647a2543537a8fda7482f92b370646c79b19777938489'), (ROOT / 'run_ufs_pci_compat.py', '4be8b5fbc9f9a7bd21cf19c317f261eae2fb7a3919ffe80e5adb6faade1061c2'), (ROOT / 'source/.github/workflows/tb371fc.yml', '6f0545c1893f7b0914c4ab5b90e6f43cb90cfa8abc708a2edb12b51fae2e3be0')]:
            raw = path.read_bytes()
            require(sha(raw) == expected, 'Reviewed source bytes changed')
            result['source_files'].append({'path': str(path), 'bytes': len(raw), 'sha256': sha(raw)})
        result['limitations'] = ['852 is independently summed from nine actual runner output lines; individual guest PASS labels reside in the separate CPU artifact and were not downloaded here.', 'Both invalid configurations have actual runner rejection lines; the job log does not include their complete captured stderr.', 'This does not qualify the board/Linux job, physical cache capacities, timing, DSU, performance or original Android boot.']
        result['passed'] = True
    except Exception:
        result['error'] = traceback.format_exc()
    with out.open('x') as stream:
        json.dump(result, stream, indent=2)
        stream.write('\n')
    print(json.dumps({'passed': result['passed'], 'receipt': str(out), 'receipt_bytes': out.stat().st_size, 'receipt_sha256': sha(out.read_bytes()), 'error': result.get('error')}))
    return 0 if result['passed'] else 1
if __name__ == '__main__':
    raise SystemExit(main())

def need(value, label):
    if not value:
        raise ValueError(label)


def exact(value, expected, label):
    need(type(value) is type(expected), label + ': type')
    if isinstance(expected, dict):
        need(set(value) == set(expected), label + ': keys')
        for key in expected:
            exact(value[key], expected[key], label + '.' + str(key))
    elif isinstance(expected, list):
        need(len(value) == len(expected), label + ': length')
        for index, (a, b) in enumerate(zip(value, expected)):
            exact(a, b, label + '.' + str(index))
    else:
        need(value == expected, label + ': value')


def unique_pairs(pairs):
    out = {}
    for key, value in pairs:
        need(key not in out, 'duplicate JSON key: ' + key)
        out[key] = value
    return out


def document(raw):
    def nonfinite(value):
        raise ValueError('nonfinite JSON number: ' + value)
    return json.loads(raw, object_pairs_hook=unique_pairs, parse_constant=nonfinite)


def read(path):
    return document(Path(path).read_bytes())


def pin(path, digest=None):
    path = Path(path)
    need(path.is_absolute() and path.is_file() and not path.is_symlink(), 'regular absolute file: ' + str(path))
    hasher = hashlib.sha256()
    count = 0
    with path.open('rb') as stream:
        while True:
            block = stream.read(1024 * 1024)
            if not block:
                break
            count += len(block)
            hasher.update(block)
    actual = {'path': str(path), 'bytes': count, 'sha256': hasher.hexdigest()}
    if digest is not None:
        exact(actual['sha256'], digest, 'complete file digest: ' + str(path))
    return actual


def spec_shape(spec):
    need(type(spec) is dict and set(spec) == {'path', 'bytes', 'sha256'}, 'complete file pin fields')
    need(type(spec['path']) is str and Path(spec['path']).is_absolute(), 'absolute pin path')
    need(type(spec['bytes']) is int and spec['bytes'] >= 0, 'pin length')
    need(type(spec['sha256']) is str and re.fullmatch('[0-9a-f]{64}', spec['sha256']), 'full lowercase SHA256')


def checked_spec(spec):
    spec_shape(spec)
    actual = pin(spec['path'], spec['sha256'])
    exact(actual, spec, 'actual complete file pin')
    return actual


def safe_relative(value):
    need(type(value) is str, 'relative path type')
    path = PurePosixPath(value)
    need(value and not path.is_absolute() and '..' not in path.parts and str(path) == value and '\x00' not in value,
         'normalized bounded relative path')
    return path

def validate_current_sources(assets):
    helper_path = assets['tree_math']['path']
    module = types.ModuleType('independent_current_git_tree_math')
    # Only this pinned, reviewed pure arithmetic helper is evaluated. No model,
    # generator, verifier, CI harness, auditor or native runner is imported.
    exec(compile(Path(helper_path).read_bytes(), helper_path, 'exec'), module.__dict__)
    entries = []
    for label, expected in PUBLICATION_EXPECTATIONS:
        entries.append((read(assets[label + '_packet']['path']), read(assets[label + '_base_trees']['path']),
                        read(assets[label + '_readback']['path']), expected))
    publications, objects = module.source_chain(entries)
    retained = read(assets['independent_source_chain']['path'])
    exact(publications, retained['publications'][:2], 'retained first two complete publication proofs')
    extra = read(assets['current_extra_trees']['path'])
    provenance = read(assets['current_extra_tree_provenance']['path'])
    need(len(extra) == len(provenance) == 8, 'eight connected complete extra directory trees')
    for row in provenance:
        actual = checked_spec(row['source'])
        raw = read(actual['path'])
        if 'result' in raw:
            raw = document(raw['result']['structuredContent']['content'])
        exact(raw, extra[row['tree_path']], 'complete retained/fresh API tree object')
        exact(raw['sha'], row['git_tree_sha1'], 'extra tree identity')
    objects = module.extend_current_objects(objects, extra, THIRD_PUBLICATION_EXPECTATION['parent_tree'])
    # The existing format is normalized from all31 pinned full raw replies;
    # the source preparation proof retains their byte and Git object checks.
    third = module.publication(read(assets['third_packet']['path']), objects,
                               read(assets['third_normalized_readback']['path']), THIRD_PUBLICATION_EXPECTATION)
    publications.append(third)
    objects = third['tree_objects']
    fourth = module.publication(read(assets['fourth_packet']['path']), objects,
                                read(assets['fourth_normalized_readback']['path']), FOURTH_PUBLICATION_EXPECTATION)
    publications.append(fourth)
    objects = fourth['tree_objects']
    exact(publications, retained['publications'], 'recomputed four-hop complete publication proof')
    source_merge = module.document_merge(objects, read(assets['source_document_merge_original_objects']['path']),
                                         read(assets['source_document_merge_readback']['path']),
                                         SOURCE_DOCUMENT_MERGE_EXPECTATION)
    objects = source_merge['tree_objects']
    exact({key: value for key, value in source_merge.items() if key != 'tree_objects'},

def validate_archive_bytes(artifact, reports):
    actual = checked_spec(artifact['archive'])
    observed = {}
    counts = {}
    for label, report in reports.items():
        rows = report.get('verified_members', report.get('verified_files'))
        if rows is None:
            continue
        if isinstance(rows, dict):
            entries = list(rows.items())
        else:
            need(type(rows) is list, 'known verified member collection')
            entries = [(row['member'], row) for row in rows]
        if 'verified_member_count' in report:
            exact(len(entries), report['verified_member_count'], 'actual audit member count')
        counts[label] = len(entries)
        for name, row in entries:
            safe_relative(name)
            expected = {key: row[key] for key in ('bytes', 'sha256')}
            need(type(expected['bytes']) is int and 0 <= expected['bytes'] <= 1024 ** 3 and
                 re.fullmatch('[0-9a-f]{64}', expected['sha256']), 'bounded verified member identity')
            if name in observed:
                exact(observed[name], expected, 'independent audit agreement on complete member')
            observed[name] = expected
    need(set(counts) == {'integration', 'ddr_stats_offset', 'rpm_stats_offset', 'spss', 'llcc', 'dispcc', 'geni_hs', 'geni_source'},
         'all eight complete member inventories retained; DP/DSI preserve separate scoped receipts')
    exact(observed[QEMU_MEMBER], {key: artifact['qemu'][key] for key in ('bytes', 'sha256')}, 'binary member agreement')
    with zipfile.ZipFile(actual['path']) as archive:
        infos = archive.infolist()
        need(len(infos) == len({info.filename for info in infos}), 'archive has no duplicate names')
        by_name = {info.filename: info for info in infos}
        for name, expected in sorted(observed.items()):
            need(name in by_name and not by_name[name].is_dir() and by_name[name].file_size == expected['bytes'],
                 'actual bounded archive member shape')
            digest = hashlib.sha256(); count = 0; prefix = b''
            with archive.open(name) as stream:
                while True:
                    raw = stream.read(1024 * 1024)
                    if not raw:
                        break
                    if not prefix:
                        prefix = raw[:20]
                    count += len(raw)
                    need(count <= expected['bytes'], 'uncompressed member exceeds reviewed complete length')
                    digest.update(raw)
            exact({'bytes': count, 'sha256': digest.hexdigest()}, expected, 'full independently reread member ' + name)
            if name == QEMU_MEMBER:
                need(len(prefix) == 20 and prefix[:6] == b'\x7fELF\x02\x01', 'actual complete ELF prefix; never executed')
    exact(pin(actual['path']), actual, 'archive unchanged after complete independent member reread')
    return {'archive': actual, 'rehashed_member_count': len(observed),
            'rehashed_uncompressed_bytes': sum(row['bytes'] for row in observed.values()),
            'independent_inventory_counts': counts, 'qemu_executed': False, 'extracted_members': 0}

def main():
    # The frozen unbound preparatory source cannot read a binding, actual ZIP,
    # receipts or private inputs. Root may create a reviewed versioned successor
    # changing only this binding digest once the actual evidence is complete.
    need(re.fullmatch('[0-9a-f]{64}', BINDING_SHA256),
         'GENI qualification binding is UNBOUND; no evidence read and no qualification written')
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--binding', type=Path, required=True)
    parser.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    need(not args.out.exists() and args.out.is_absolute(), 'new absolute qualification output required')
    binding_pin = pin(args.binding, BINDING_SHA256)
    result = assemble(read(args.binding), binding_pin)
    raw = (json.dumps(result, indent=2, sort_keys=True, allow_nan=False) + '\n').encode()
    with args.out.open('xb') as stream:
        stream.write(raw)
    print(json.dumps({'passed': True, 'qualification': pin(args.out), 'native_executed': False}))
    return 0

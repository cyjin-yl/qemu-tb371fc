class AuditFailure(RuntimeError):
    pass

def need(ok, message):
    if not ok:
        raise AuditFailure(message)

def same(actual, expected, where):
    """Type-sensitive comparison; bool is never accepted as an integer."""
    need(type(actual) is type(expected), f"{where}: type {type(actual).__name__} != {type(expected).__name__}")
    if type(expected) is dict:
        need(set(actual) == set(expected), f"{where}: object keys differ")
        for key in expected:
            same(actual[key], expected[key], where + "." + str(key))
    elif type(expected) in (tuple, list):
        need(len(actual) == len(expected), f"{where}: length {len(actual)} != {len(expected)}")
        for i, value in enumerate(expected):
            same(actual[i], value, f"{where}[{i}]")
    else:
        need(actual == expected, f"{where}: {str(actual)[:160]} != {str(expected)[:160]}")

def uint(value, bits, where, signed=False):
    need(type(value) is int, where + ": integer required")
    low, high = (-(1 << (bits - 1)), 1 << (bits - 1)) if signed else (0, 1 << bits)
    need(low <= value < high, where + ": integer out of range")
    return value

def boolean(value, where):
    need(type(value) is bool, where + ": boolean required")
    return value

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def file_hash(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for block in iter(lambda: f.read(CHUNK), b""):
            h.update(block)
    return h.hexdigest()

def pairs_unique(pairs):
    result = {}
    for key, value in pairs:
        need(key not in result, "Duplicate JSON key " + key)
        result[key] = value
    return result

def json_value(raw, where):
    def bad_constant(value):
        raise AuditFailure(where + ": nonfinite JSON " + value)
    try:
        return json.loads(raw, object_pairs_hook=pairs_unique, parse_constant=bad_constant)
    except (ValueError, UnicodeError) as error:
        raise AuditFailure(where + ": invalid JSON: " + str(error)) from error

def safe_name(name):
    need(type(name) is str and name and "\\" not in name and "\x00" not in name,
         "Invalid archive path")
    p = PurePosixPath(name)
    need(not p.is_absolute() and all(part not in ("", ".", "..") for part in name.split("/")),
         "Unsafe archive path " + name)
    return name

class Store:
    """One ZIP or existing directory; paths are never extracted or executed."""
    def __init__(self, zip_path=None, directory=None):
        self.zip_path = Path(zip_path).resolve() if zip_path else None
        self.root = Path(directory).resolve() if directory else None
        self.z = ZipFile(self.zip_path) if self.zip_path else None
        self.entries, self.observed = {}, {}
        if self.z:
            self.initial = self.zip_path.stat()
            for entry in self.z.infolist():
                name = entry.filename.rstrip("/")
                if not name:
                    continue
                safe_name(name)
                need(name not in self.entries, "Duplicate ZIP member " + name)
                need(not stat.S_ISLNK(entry.external_attr >> 16), "Symlink ZIP member " + name)
                if not entry.is_dir():
                    self.entries[name] = entry
        else:
            need(self.root.is_dir(), "Result directory does not exist")
            for p in self.root.rglob("*"):
                need(not p.is_symlink(), "Symlink in result directory: " + str(p))
                if p.is_file():
                    self.entries[p.relative_to(self.root).as_posix()] = p
        need(len(self.entries) < 200000, "Unreasonable member count")

    def size(self, name):
        safe_name(name)
        need(name in self.entries, "Missing member " + name)
        entry = self.entries[name]
        return entry.file_size if self.z else entry.stat().st_size

    def open(self, name):
        self.size(name)
        return self.z.open(self.entries[name]) if self.z else self.entries[name].open("rb")

    def read(self, name, limit=MAX_SMALL):
        need(self.size(name) <= limit, "Member too large for bounded small-file read: " + name)
        with self.open(name) as f:
            raw = f.read(limit + 1)
        same(len(raw), self.size(name), name + ".actual size")
        self.observed[name] = {"bytes": len(raw), "sha256": sha(raw)}
        return raw

    def text(self, name):
        return self.read(name).decode("utf-8", errors="strict")

    def json(self, name):
        return json_value(self.read(name), name)

    def finish(self):
        if self.z:
            now = self.zip_path.stat()
            same((now.st_dev, now.st_ino, now.st_size, now.st_mtime_ns),
                 (self.initial.st_dev, self.initial.st_ino, self.initial.st_size, self.initial.st_mtime_ns),
                 "ZIP changed during audit")
            self.z.close()

def one_bit_difference(store, original, changed, expected_absolute, expected_xor, source_sha, changed_sha):
    same(store.size(original), store.size(changed), changed + ".whole stream length")
    offset, differences = 0, []
    a_hash, b_hash = hashlib.sha256(), hashlib.sha256()
    with store.open(original) as a, store.open(changed) as b:
        while True:
            left, right = a.read(CHUNK), b.read(CHUNK)
            same(len(left), len(right), changed + ".stream chunk length")
            if not left:
                break
            a_hash.update(left)
            b_hash.update(right)
            if left != right:
                for i, (x, y) in enumerate(zip(left, right)):
                    if x != y:
                        differences.append((offset + i, x, y))
                        need(len(differences) == 1, changed + ": more than one actual changed byte")
            offset += len(left)
    same(a_hash.hexdigest(), source_sha, original + ".unmodified after copying")
    same(b_hash.hexdigest(), changed_sha, changed + ".stable actual mutated bytes")
    same(len(differences), 1, changed + ".actual differing byte count")
    at, before, after = differences[0]
    same(at, expected_absolute, changed + ".actual changed byte offset")
    same(before ^ after, expected_xor, changed + ".actual one-bit xor")
    same((before ^ after).bit_count(), 1, changed + ".actual changed bit count")
    return {"byte_offset": at, "before": before, "after": after, "changed_bits": 1,
            "all_other_stream_bytes_equal": True, "original_stream_sha256_unchanged": True}

def qmp_calls(store, directory):
    path, calls, pending, last_id = directory + "/qmp.jsonl", [], None, 0
    greeting = 0
    for line_number, line in enumerate(store.text(path).splitlines(), 1):
        row = json_value(line, f"{path}:{line_number}")
        need(type(row) is dict and len(row) == 1, path + ": malformed QMP log entry")
        if "send" in row:
            need(pending is None, path + ": overlapping QMP sends")
            request = row["send"]
            need(type(request) is dict and type(request.get("execute")) is str, path + ": malformed QMP command")
            same(request.get("id"), last_id + 1, path + ": sequential request id")
            last_id += 1
            pending = {"request": request, "response": None, "line": line_number}
            calls.append(pending)
        else:
            need("receive" in row and type(row["receive"]) is dict, path + ": malformed QMP receive")
            response = row["receive"]
            if "QMP" in response:
                greeting += 1
                need(not calls, path + ": late QMP greeting")
                continue
            if "event" in response:
                continue
            need(pending is not None, path + ": unsolicited QMP response")
            same(response.get("id"), pending["request"]["id"], path + ": unmatched response id")
            need(("return" in response) != ("error" in response), path + ": malformed QMP verdict")
            pending["response"] = response
            pending = None
    same(greeting, 1, path + ".greeting count")
    # A failed incoming migration can close the socket before its reply.
    if pending is not None:
        same(pending["request"]["execute"], "migrate-incoming", path + ": truncated non-migration QMP request")
    return calls

def operation(call):

def reply(call):
    response = call["response"]
    need(type(response) is dict and "return" in response and "error" not in response,
         "Missing successful QMP response for " + operation(call))
    return response["return"]

    channels = args["channels"]

    addr = channel.get("addr")
    need(type(addr) is dict and addr.get("transport") == "file", "Migration is not from a local file")
    same(addr.get("offset"), 0, "Required file migration offset starts at byte zero")
    filename = addr.get("filename")
    need(type(filename) is str and PurePosixPath(filename).name == expected,
         "Actual migration filename differs from reviewed stream")

def qtest_time(calls, where):
    now = 0
    for item in calls:
        same(item["virtual_ns_before"], now, where + ": local virtual-clock sequence")
        command = item["command"].split()
        need(command, where + ": empty QTest command")
        if command[0] in ("clock_set", "clock_step"):
            need(len(command) == 2, where + ": unbounded clock command")
            value = int(command[1], 0)
            need(value >= 0, where + ": negative virtual time")
            now = value if command[0] == "clock_set" else now + value
            parts = item["reply"].split()
            need(len(parts) == 2, where + ": malformed clock response")
            same(int(parts[1], 0), now, where + ": clock response value")
    return now

def hash_member(store, name):
    digest, size = hashlib.sha256(), 0
    with store.open(name) as source:
        for chunk in iter(lambda: source.read(CHUNK), b""):
            digest.update(chunk)
            size += len(chunk)
    same(size, store.size(name), name + ".full bytes")
    result = {"bytes": size, "sha256": digest.hexdigest()}
    store.observed[name] = result
    return result

def fields_equal(actual, expected, where):
    need(type(actual) is dict, where + ": object required")
    for key, value in expected.items():
        need(key in actual, where + ": missing " + key)
        same(actual[key], value, where + "." + key)

def completed_polls(calls, saved_reply, where):
    need(calls, where + ": no migration status response")
    for index, call in enumerate(calls):
        same(operation(call), "query-migrate", where + ".status polling only")
        need("arguments" not in call["request"], where + ".status arguments")
        status = reply(call)
        need(type(status) is dict and type(status.get("status")) is str, where + ".actual status")
        need(status["status"] not in ("failed", "cancelled"), where + ".failed migration")
        same(status["status"] == "completed", index == len(calls) - 1,
             where + ".first completed poll terminates polling")
    same(reply(calls[-1]), saved_reply, where + ".saved versus actual completion response")

def locate_case(store, name):
    candidates = [p for p in store.entries
                  if re.search(r"(?:^|/)components/[0-9]+-" + re.escape(name) + r"/case\.json$", p)]
    same(len(candidates), 1, "Exactly one compiled case " + name)
    path = candidates[0]
    root = path.rsplit("/", 1)[0]
    case = store.json(path)
    fields_equal(case, {"name": name, "passed": True, "exit_code": 0}, name + ".actual harness verdict")
    need(not any(k in case for k in ("error", "artifact_error", "timed_out")), name + ": failed/incomplete case")
    need(type(case.get("completed_utc")) is str and case["completed_utc"], name + ": missing completion time")
    logs = {}
    for filename in ("stdout.log", "stderr.log"):
        raw = store.read(root + "/" + filename)
        same(case["log_sha256"][filename], sha(raw), name + ".actual harness log hash " + filename)
        logs[filename] = raw
    command = case["command"]
    need(type(command) is list and all(type(s) is str for s in command), name + ".actual harness argv")
    need(not set(command).intersection(("-O", "-OO", "-c")), name + ": assertion-disabled or inline producer")
    return path, root, case, logs

    return path

def raw_registers(value):
    raw = bytes.fromhex(value['registers-hex'])
    return [int.from_bytes(raw[i:i+4],'big') for i in range(0,16384,4)]


def irq_from_state(value):
    r = raw_registers(value)
    rxwords = (value['rx-count']+3)//4
    secondary = r[0x640//4] | (RX_WM if rxwords and rxwords >= r[0x810//4] else 0)
    primary = r[0x610//4] | (TX_WM if r[0x80c//4] and
                           (value['tx-count']+3)//4 < r[0x80c//4] else 0)
    primary |= secondary & RX_WM
    if secondary & r[0x644//4]:
        primary |= 1<<31
    available = (not value['reset-held'] and all(value[k]>0 for k in
                 ('se-clock-period','m-ahb-clock-period','s-ahb-clock-period')))
    return bool(available and (((r[0xe1c//4]&4) and (primary&r[0x614//4])) or
                              ((r[0xe1c//4]&8) and (secondary&r[0x644//4]))))


def check_engine(value, where, hs=True):
    same(set(value),set(DYNAMIC),where+'.17 pure fields')
    for name in DUINT:
        uint(value[name],64,where+'.'+name)
    for name in DBOOL:
        boolean(value[name],where+'.'+name)
    for name,size in (('registers-hex',16384),('tx-fifo-hex',64),('rx-fifo-hex',64)):
        need(type(value[name]) is str and re.fullmatch('[0-9a-f]{'+str(size*2)+'}',value[name]),
             where+'.canonical complete hex '+name)
    same(value['fifo-bytes'],64,where+'.fixed FIFO geometry')
    need(0<=value['tx-count']<=64 and 0<=value['rx-count']<=64,where+'.FIFO counts')
    need(value['tx-count']<=value['tx-remaining'],where+'.queued vs remaining')
    need(not value['tx-active'] or value['tx-remaining']>0,where+'.active TX')
    need(value['tx-active'] or not (value['tx-count'] or value['tx-remaining']),where+'.inactive TX')
    need(not (value['tx-zero-pending'] and value['tx-active']),where+'.zero command consistency')
    need(not value['reset-held'] or not any(value[k] for k in
         ('tx-active','rx-active','tx-count','rx-count','tx-zero-pending')),where+'.held reset state')
    same(value['irq-level'],irq_from_state(value),where+'.independent pure IRQ formula')
    if hs:
        need(value['tx-sent']+value['tx-remaining']<=0xffffff,where+'.24-bit command accounting')
        regs=raw_registers(value)
        for i,word in enumerate(regs):
            off=i*4
            mask=WRITE_MASKS.get(off,0) if off not in NONSTORED else 0
            if off==0x610:mask=0x2b00003f
            if off==0x640:mask=0x0b00003f
            need(not (word&~mask),where+f'.raw register {off:#x} reserved bits')
            if off in (0x600,0x630):need(word in (0,START),where+'.stored command')
            if off in (0x268,0x28c):same(word,8,where+'.eight-bit words')
            if off in (0x260,0x284):need(word in (0,0x4380e),where+'.packing0')
            if off in (0x264,0x288):need(word in (0,0xc3e0e),where+'.packing1')
            if off==0x254:same(word,0,where+'.byte granularity')
        if value['tx-active'] or value['tx-zero-pending']:
            same(regs[0x600//4],START,where+'.active TX command')
        if value['rx-active']:same(regs[0x630//4],START,where+'.active RX command')
        if value['tx-zero-pending']:same(value['tx-sent'],0,where+'.zero command sent count')
    return value


def check_world(value,enabled,where):
    same(set(value),{'hs','identity','legacy'},where+'.world keys')
    same(value['identity'],identity(enabled),where+'.independent immutable reference')
    check_engine(value['hs'],where+'.hs')
    check_engine(value['legacy'],where+'.legacy',False)
    return value


def qom_queries(value):
    return ([({'path':DEVICE,'property':k},value['hs'][k]) for k in DYNAMIC]+
            [({'path':DEVICE,'property':k},value['identity'][k]) for k in IFIELDS]+
            [({'path':LEGACY,'property':k},value['legacy'][k]) for k in DYNAMIC])


def match_world(calls,at,value,where):
    expected=qom_queries(value)
    need(at+49<=len(calls),where+'.complete49 getter block')
    for i,(args,val) in enumerate(expected):
        same(operation(calls[at+i]),'qom-get',where+f'.getter{i}')
        same(calls[at+i]['request'].get('arguments'),args,where+f'.getter arguments{i}')
        same(reply(calls[at+i]),val,where+f'.actual getter value{i}')
    return at+49


def decode_world(calls,at,enabled,where):
    names=[(DEVICE,k,'hs') for k in DYNAMIC]+[(DEVICE,k,'identity') for k in IFIELDS]+[(LEGACY,k,'legacy') for k in DYNAMIC]
    value={'hs':{},'identity':{},'legacy':{}}
    need(at+49<=len(calls),where+'.complete world')
    for i,(path,name,part) in enumerate(names):
        call=calls[at+i]
        same(operation(call),'qom-get',where+'.getter operation')
        same(call['request'].get('arguments'),{'path':path,'property':name},where+'.getter order')
        value[part][name]=reply(call)
    return check_world(value,enabled,where)


def audit_all_qom(calls,enabled,where,allow_set=False):
    worlds=[];engines=[];i=0
    while i<len(calls):
        call=calls[i];op=operation(call)
        if op=='qom-get':
            args=call['request'].get('arguments',{});same(set(args),{'path','property'},where+'.QOM arguments')
            path,name=args['path'],args['property'];need(path in (DEVICE,LEGACY),where+'.QOM path')
            need(name in DYNAMIC or path==DEVICE and name in IFIELDS,where+'.QOM property')
            val=reply(call)
            if path==DEVICE and name in IFIELDS:same(val,identity(enabled)[name],where+'.identity getter '+name)
            elif name in DUINT:uint(val,64,where+'.'+name)
            elif name in DBOOL:boolean(val,where+'.'+name)
            else:
                n=32768 if name=='registers-hex' else 128
                need(type(val) is str and re.fullmatch('[0-9a-f]{'+str(n)+'}',val),where+'.QOM complete hex')
            if name==DYNAMIC[0] and i+17<=len(calls):
                block=calls[i:i+17]
                if all(operation(c)=='qom-get' and c['request'].get('arguments')=={'path':path,'property':n}
                       for c,n in zip(block,DYNAMIC)):
                    value={n:reply(c) for n,c in zip(DYNAMIC,block)}
                    check_engine(value,where+f'.engine-block{i}',path==DEVICE)
                    engines.append({'index':i,'path':path,'value':value})
                    if path==DEVICE and i+49<=len(calls) and calls[i+17]['request'].get('arguments')=={'path':DEVICE,'property':IFIELDS[0]}:
                        worlds.append({'index':i,'value':decode_world(calls,i,enabled,where+f'.world{i}')})
        elif op=='qom-set':
            need(allow_set,where+'.unexpected QOM mutation request')
            need(type(call['response']) is dict and 'error' in call['response'],where+'.readonly QOM rejection')
        elif op=='qom-list':
            path=call['request'].get('arguments',{}).get('path')
            need(path in (DEVICE,DEVICE+'/se-clk',DEVICE+'/m-ahb',DEVICE+'/s-ahb'),where+'.QOM list scope')
            props=reply(call);need(type(props) is list,where+'.QOM property list')
            names=[p.get('name') for p in props];same(len(names),len(set(names)),where+'.unique QOM names')
            need(not any(n.startswith('qtest-') for n in names),where+'.no fixture QOM properties')
            if path==DEVICE:
                types={p['name']:p['type'] for p in props}
                for n in DSTR:same(types.get(n),'string',where+'.string property')
                for n in DUINT+IUINT:same(types.get(n),'uint64',where+'.uint64 property')
                for n in DBOOL+IBOOL:same(types.get(n),'bool',where+'.bool property')
        else:
            need(op in ('qmp_capabilities','cont','stop','query-status','human-monitor-command','system_reset',
                        'migrate','migrate-incoming','query-migrate','quit'),where+'.unreviewed QMP operation '+op)
            if op=='human-monitor-command':
                same(call['request'].get('arguments'),{'command-line':'info mtree -f'},where+'.read-only bounded HMP command')
            elif op not in ('migrate','migrate-incoming'):
                need('arguments' not in call['request'],where+'.no extra control arguments')
            if op!='migrate-incoming':reply(call)
        i+=1
    return {'worlds':worlds,'engines':engines}


def qtest_trace(store,directory):
    path=directory+'/qtest.jsonl';calls=[];pending=None;irq=[];level=False
    for line_number,line in enumerate(store.text(path).splitlines(),1):
        row=json_value(line,path+f':{line_number}')
        if 'send' in row:
            same(set(row),{'send','virtual_ns_before'},path+'.send keys')
            need(pending is None and type(row['send']) is str,path+'.overlapping command')
            pending={'command':row['send'],'virtual_ns_before':row['virtual_ns_before'],
                     'reply':None,'line':line_number,'irq_before':level,'irq_after':None}
            calls.append(pending)
        else:
            same(set(row),{'receive'},path+'.reply keys');text=row['receive']
            need(type(text) is str,path+'.text reply')
            if text.startswith('IRQ '):
                m=re.fullmatch(r'IRQ (raise|lower) ([0-9]+)',text);need(m is not None,path+'.IRQ syntax')
                same(int(m[2]),0,path+'.only HS sysbus line0')
                value=m[1]=='raise';need(value!=level,path+'.genuine IRQ transition')
                level=value;irq.append({'level':m[1],'number':0,'line':line_number})
                continue
            need(pending is not None and (text=='OK' or text.startswith('OK ')),path+'.successful QTest response')
            pending['reply']=text;pending['irq_after']=level;pending=None
    need(pending is None,path+'.truncated command')
    qtest_time(calls,path)
    return calls,irq


def backend_trace(store,directory):
    path=directory+'/backend.jsonl';rows=[];sent={'hs':b'','legacy':b''};received={'hs':b'','legacy':b''}
    connections=[];starts=[];finishes=[];send_count=recv_count=0
    for number,line in enumerate(store.text(path).splitlines(),1):
        row=json_value(line,path+f':{number}');need(type(row) is dict,path+'.object entry')
        rows.append(row)
        if 'send' in row or 'receive' in row:
            action='send' if 'send' in row else 'receive'
            same(set(row),{action,'bytes','engine'},path+'.data record')
            engine=row['engine'];need(engine in sent,path+'.backend engine')
            data=row[action];need(type(data) is str and re.fullmatch('(?:[0-9a-f]{2})+',data),path+'.exact peer bytes')
            raw=bytes.fromhex(data);same(row['bytes'],len(raw),path+'.data byte count')
            (sent if action=='send' else received)[engine]+=raw
            if action=='send':send_count+=1
            else:recv_count+=1
        else:
            event=row.get('event')
            if event=='connected':
                same(set(row),{'event','engine'},path+'.connection entry');connections.append(row['engine'])
            elif event=='no-output':
                same(set(row),{'event','engine','timeout_seconds','consumed_bytes'},path+'.nonconsuming probe')
                need(row['engine'] in sent,path+'.probe engine');same(row['consumed_bytes'],0,path+'.probe does not consume')
                need(type(row['timeout_seconds']) in (int,float) and 0<=row['timeout_seconds']<=5,path+'.bounded output probe')
            elif event=='failure-observation-start':starts.append(row)
            elif event=='failure-observation-complete':finishes.append(row)
            else:raise AuditFailure(path+': unreviewed backend event '+str(event))
    return {'rows':rows,'sent':sent,'received':received,'connections':connections,
            'starts':starts,'finishes':finishes,'send_count':send_count,'receive_count':recv_count}


def scalar(call):
    parts=call['reply'].split();same(len(parts),2,'scalar QTest reply')
    return int(parts[1],0)


def parts(call):
    words=call['command'].split();need(words,'nonempty QTest command')
    if words[0] in ('irq_intercept_out','set_irq_in'):return tuple(words)
    return (words[0],)+tuple(int(x,0) for x in words[1:])


def memory_call(call):
    return call['command'].split()[0] in ('readb','readw','readl','readq','writeb','writew','writel','writeq','read','write')


def memory_counts(calls):
    mem=[c for c in calls if memory_call(c)]
    dev=sum((HS_BASE<=parts(c)[1]<HS_BASE+0x4000 or OLD_BASE<=parts(c)[1]<OLD_BASE+0x4000 or
             WRAPPER<=parts(c)[1]<WRAPPER+4 or GCC<=parts(c)[1]<GCC+0x1f0000) for c in mem)
    ram=sum(RAM<=parts(c)[1]<RAM+4 for c in mem)
    return {'memory':len(mem),'device_mmio':dev,'ram':ram}

def diagnostic_rows(text):
    core=re.findall(r"Invalid (read|write) at addr 0x([0-9a-fA-F]+), size ([0-9]+), region '([^']+)', reason: ([^\r\n]+)",text)
    same(len(core),text.count('Invalid read at addr ')+text.count('Invalid write at addr '),'all generic rejections parsed')
    core=[(op,int(a,16),int(n),region,reason) for op,a,n,region,reason in core]
    pattern=re.escape(TYPE)+r': rejected (wrapper )?(read|write) offset=0x([0-9a-f]+) size=([0-9]+)(?: value=0x([0-9a-f]+))? reason=([^\r\n]+)'
    device=re.findall(pattern,text)
    same(len(device),text.count(TYPE+': rejected '),'all device rejections parsed')
    device=[(bool(w),op,int(a,16),int(n),int(v,16) if v else None,reason) for w,op,a,n,v,reason in device]
    for wrapper,op,off,size,value,reason in device:
        same(size,4,'delivered callback width')
        need(off%4==0,'delivered aligned callback')
        same(value is None,op=='read','write diagnostic contains its actual value')
        need(reason in ('unsupported-register','read-only','reserved-bits','unsupported-dma','unsupported-format','bus-unavailable'),
             'bounded callback rejection reason')
    return core,device

def scan_uart_sections(store,path,expected_world):
    payloads=dict(zip((TYPE,'qcom-geni-uart'),wire_from_world(expected_world)))
    needles={name:bytes((len(name),))+name.encode()+b'\x00\x00\x00\x00\x00\x00\x00\x01' for name in payloads}
    found={name:{} for name in payloads};tail=b'';total=0;hasher=hashlib.sha256();magic=None
    with store.open(path) as stream:
        for chunk in iter(lambda:stream.read(CHUNK),b''):
            if magic is None:magic=chunk[:8]
            hasher.update(chunk);block=tail+chunk;start=total-len(tail)
            for name,needle in needles.items():
                pos=0
                while True:
                    at=block.find(needle,pos)
                    if at<0:break

def expected_source_world(enabled):

class QtCursor:
    def __init__(self,fixture):self.f=fixture;self.i=0;self.expected_irqs=[]
    def take(self,expected,value=None):
        need(self.i<len(self.f.qt),self.f.path+'.missing QTest operation')
        row=self.f.qt[self.i];self.i+=1;same(parts(row),expected,self.f.path+f'.QTest operation{self.i}')
        if value is not None:same(scalar(row),value,self.f.path+f'.QTest value{self.i}')
        elif expected[0].startswith('write') or expected[0] in ('irq_intercept_out','set_irq_in'):
            same(row['reply'],'OK',self.f.path+'.write/control acknowledgment')
        return row
    def write(self,address,value):return self.take(('writel',address,value))
    def read(self,address,value=None):return self.take(('readl',address),value)
    def gate(self,address,mask,enabled):
        row=self.read(address);old=scalar(row)
        self.write(address,old|mask if enabled else old&~mask)
    def configure(self,cfg=1):
        for off,val in ((0x17738,0),(0x1773c,0),(0x17740,0),(0x17734,cfg),(0x17730,1)):
            self.write(GCC+off,val)
        n=0
        while True:
            n+=1;need(n<=100,self.f.path+'.bounded RCG settling')
            self.take(('clock_step',1000));row=self.read(GCC+0x17730)
            if scalar(row)&1==0:break
        return n
    def bring_up(self):self.configure();self.gate(HS_VOTE,SE|M_AHB|S_AHB,True)
    def packing(self):
        for off,val in ((0x254,0),(0x260,0x4380e),(0x264,0xc3e0e),(0x284,0x4380e),(0x288,0xc3e0e),(0x268,8),(0x28c,8)):
            self.write(HS_BASE+off,val);self.read(HS_BASE+off,val)
    def irq(self,value):
        row=self.take(('endianness',));same(row['reply'],'OK little',self.f.path+'.read-only IRQ protocol fence')
        same(row['irq_after'],value,self.f.path+'.actual device IRQ output')
        self.expected_irqs.append(value)
    def flags(self,address,ones=0,zeros=0):
        value=scalar(self.read(address));need((value&ones)==ones and not(value&zeros),self.f.path+'.actual status bits')
        return value
    def finish(self):same(self.i,len(self.f.qt),self.f.path+'.all QTest commands consumed')

def deleted_span_difference(store,original,changed,offset,length,source_sha,changed_sha):
    same(store.size(original)-length,store.size(changed),changed+'.exact deleted length')
    with store.open(original) as a,store.open(changed) as b:
        remaining=offset
        while remaining:
            n=min(CHUNK,remaining);same(a.read(n),b.read(n),changed+'.unaltered prefix');remaining-=n
        same(len(a.read(length)),length,changed+'.complete removed guard')
        while True:
            left,right=a.read(CHUNK),b.read(CHUNK);same(left,right,changed+'.unaltered complete suffix')
            if not left:break
    same(hash_member(store,original)['sha256'],source_sha,changed+'.original remains exact')
    same(hash_member(store,changed)['sha256'],changed_sha,changed+'.actual removed-guard stream hash')
    return {'removed_offset':offset,'removed_bytes':length,'all_remaining_bytes_equal':True,
            'guard_removed_inside_present_section':True,'entire_missing_device_section_tested':False}

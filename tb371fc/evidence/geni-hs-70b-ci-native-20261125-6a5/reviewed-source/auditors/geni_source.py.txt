def need(value, label):
    if not value:
        raise ValueError(label)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def blob(raw):
    return hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()


def canonical(value):
    return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()


def equal(left, right, label):
    need(canonical(left) == canonical(right), label)


def unique_pairs(items):
    value = {}
    for key, item in items:
        need(key not in value, 'duplicate JSON key: ' + key)
        value[key] = item
    return value


def document(raw):
    def reject(value):
        raise ValueError('nonfinite JSON: ' + value)
    return json.loads(raw, object_pairs_hook=unique_pairs, parse_constant=reject)


def identity(raw, pin, label):
    need(len(raw) == pin['bytes'] and sha(raw) == pin['sha256'] and
         ('git_blob_sha1' not in pin or blob(raw) == pin['git_blob_sha1']), label + ' complete identity differs')


def without_comments(text):
    return re.sub(r'/\*.*?\*/|//[^\n]*', '', text, flags=re.S)


def single(pattern, text, label, flags=0):
    values = list(re.finditer(pattern, text, flags))
    need(len(values) == 1, label + ': missing or duplicate occurrence')
    return values[0]


def block_after(pattern, text, label):
    text = without_comments(text)
    match = single(pattern, text, label)
    opening = text.rfind('{', match.start(), match.end())
    need(opening >= match.start() and not text[opening+1:match.end()].strip(), label + ': block delimiter')
    start = opening + 1
    depth, quoted, escaped = 1, None, False
    for index in range(start, len(text)):
        c = text[index]
        if quoted:
            if escaped:
                escaped = False
            elif c == '\\':
                escaped = True
            elif c == quoted:
                quoted = None
        elif c in ('"', "'"):
            quoted = c
        elif c == '{':
            depth += 1
        elif c == '}':
            depth -= 1
            if depth == 0:
                return text[start:index]
    raise ValueError(label + ': unterminated block')


def function_body(text, name):
    return block_after(r'(?m)^[ \t]*(?:static\s+)?[A-Za-z_][\w\s*]*?\b' +
                       re.escape(name) + r'\s*\([^;{}]*\)\s*\{', text, 'function ' + name)


def ordered_tokens(text, tokens, label):
    text = re.sub(r'\s+', '', without_comments(text))
    last = -1
    for token in tokens:
        token = re.sub(r'\s+', '', token)
        positions = [m.start() for m in re.finditer(re.escape(token), text)]
        need(positions and positions[0] > last, label + ': first-token/order ' + token)
        last = positions[0]


class IntegerDefinitions:
    """Small AST evaluator for literal C integer macros, with no eval/import."""
    def __init__(self, text=''):
        text = without_comments(text).replace('\\\n', '')
        self.values = {}
        self.expressions = {}
        for name, expression in re.findall(r'^\s*#\s*define\s+([A-Za-z_]\w*)[ \t]+([^\n]+)', text, re.M):
            self.expressions.setdefault(name, []).append(expression.strip())

    def get(self, name, visiting=()):
        need(name not in visiting, 'recursive C integer')
        need(name in self.expressions and len(self.expressions[name]) == 1, 'C macro missing/ambiguous: ' + name)
        return self.expression(self.expressions[name][0], visiting + (name,))

    def expression(self, text, visiting=()):
        text = re.sub(r'\b(0x[0-9a-fA-F]+|[0-9]+)[uUlL]+\b', r'\1', text)
        node = ast.parse(text.strip(), mode='eval').body
        binary = {ast.Add: operator.add, ast.Sub: operator.sub, ast.Mult: operator.mul,
                  ast.LShift: operator.lshift, ast.RShift: operator.rshift,
                  ast.BitOr: operator.or_, ast.BitAnd: operator.and_, ast.BitXor: operator.xor}
        def walk(value):
            if isinstance(value, ast.Constant) and type(value.value) is int:
                return value.value
            if isinstance(value, ast.Name):
                return self.get(value.id, visiting)
            if isinstance(value, ast.UnaryOp) and isinstance(value.op, (ast.USub, ast.UAdd, ast.Invert)):
                x = walk(value.operand)
                return -x if isinstance(value.op, ast.USub) else +x if isinstance(value.op, ast.UAdd) else ~x
            if isinstance(value, ast.BinOp):
                left, right = walk(value.left), walk(value.right)
                if isinstance(value.op, ast.Div):
                    need(right != 0 and left % right == 0, 'nonintegral C division')
                    return left // right
                need(type(value.op) in binary, 'unsupported C integer operator')
                if isinstance(value.op, (ast.LShift, ast.RShift)):
                    need(0 <= right <= 63, 'C shift outside bounded width')
                return binary[type(value.op)](left, right)
            if isinstance(value, ast.Call) and isinstance(value.func, ast.Name) and not value.keywords:
                args = [walk(item) for item in value.args]
                if value.func.id in ('BIT', 'BIT_ULL') and len(args) == 1:
                    need(0 <= args[0] <= 63, 'BIT width'); return 1 << args[0]
                if value.func.id in ('GENMASK', 'GENMASK_ULL') and len(args) == 2:
                    high, low = args
                    need(0 <= low <= high <= 63, 'GENMASK bounds')
                    return ((1 << (high-low+1)) - 1) << low
            raise ValueError('unsupported C integer expression')
        return walk(node)


def enumeration(text, name=None):
    pattern = r'\benum\s+' + re.escape(name) + r'\s*\{' if name else r'\benum\s*\{'
    body = block_after(pattern, text, 'enum ' + str(name))
    result, next_value = {}, 0
    for declaration in body.split(','):
        declaration = declaration.strip()
        if not declaration:
            continue
        match = re.fullmatch(r'([A-Za-z_]\w*)(?:\s*=\s*(.+))?', declaration)
        need(match and match[1] not in result, 'enum duplicate/invalid member')
        value = next_value if match[2] is None else IntegerDefinitions().expression(match[2])
        result[match[1]] = value
        next_value = value + 1
    return result


def fields(body):
    result, start, depth, quoted = [], 0, 0, False
    for i, c in enumerate(body):
        if c == '"':
            quoted = not quoted
        if quoted:
            continue
        if c in '({':
            depth += 1
        elif c in ')}':
            depth -= 1
        elif c == ',' and depth == 0:
            result.append(body[start:i].strip()); start = i + 1
    need(depth == 0 and not quoted, 'initializer nesting')
    if body[start:].strip():
        result.append(body[start:].strip())
    return result


def dt_property(body, name):
    return single(r'(?m)^\s*' + re.escape(name) + r'\s*=\s*([^;]+);', body, 'DT property ' + name)[1].strip()


def dt_cells(body, name):
    value = dt_property(body, name)
    parts = re.findall(r'<([^<>]+)>', value)
    need(parts and re.fullmatch(r'(?:\s*<[^<>]+>\s*,?\s*)+', value), 'DT cell syntax ' + name)
    return [part.split() for part in parts]


def dt_node(text, label):
    return block_after(r'\b' + re.escape(label) + r'\s*:\s*[^{}]+\{', text, 'DT node ' + label)


def dt_reg(body):
    cells = dt_cells(body, 'reg')
    need(len(cells) == 1 and len(cells[0]) == 2, 'DT single address/size pair')
    return [int(value, 0) for value in cells[0]]

def independent_inc(constants, order):
    lines = ['/* SPDX-License-Identifier: GPL-2.0-or-later */',
             '/* Generated by tb371fc/scripts/gen_geni_hs_sm8250.py; do not edit. */',
             '/* 19 full public inputs plus explicit policy and sanitized private facts. */',
             '/* Reference identity only: physical SE6 firmware/reset state is unknown. */']
    for name in order:
        kind = 'UINT64_C' if name in ('GENI_HS_SE_BASE', 'GENI_HS_WRAPPER_VERSION_ADDRESS') else 'UINT32_C'
        lines.append('#define %s %s(0x%08x)' % (name, kind, constants[name]))
    return ('\n'.join(lines) + '\n').encode()


def expected_mutants(contract, raw_inputs, references, sources, policy, facts, generated):
    """Construct complete expected bytes independently from frozen operations."""
    result = []
    def append(category, name, raw, original=None, parser_rejected=False):
        index = len(result) + 1
        result.append({'category': category, 'name': name,
                       'mutant_path': f'negative/{category}/{index:03d}-{name}.input',
                       'raw': raw, 'original': original, 'independent_semantic_rejection': parser_rejected})
    for pin in contract['public_inputs']:
        original = raw_inputs[pin['id']]; value = bytearray(original); value[len(value)//2] ^= 1
        append('complete-input-identity', pin['id'], bytes(value), original)
    for pin in contract['reference_inputs']:
        original = references[pin['role']]; value = bytearray(original); value[len(value)//2] ^= 1
        append('reference-input-identity', pin['role'], bytes(value), original)
    for source_id, pattern, replacement in contract['parser_mutations']:
        changed, count = re.subn(pattern, replacement, sources[source_id])
        need(count == 1 and changed != sources[source_id], 'one semantic mutation ' + source_id)
        alternate = dict(sources); alternate[source_id] = changed
        rejected = False
        try: independent_primary(alternate, policy, facts, contract)
        except (ValueError, KeyError): rejected = True
        need(rejected, 'independent parser accepted ' + source_id)
        append('parser-contract', source_id, changed.encode(), raw_inputs[source_id], True)
    for name, owner, path, replacement in contract['scope_mutations']:
        p, f = copy.deepcopy(policy), copy.deepcopy(facts)
        target = p if owner == 'policy' else f
        for key in path[:-1]: target = target[key]
        target[path[-1]] = replacement
        rejected = False
        try: independent_primary(sources, p, f, contract)
        except (ValueError, KeyError): rejected = True
        need(rejected, 'independent scope checker accepted ' + name)
        append('reference-scope', name, canonical({'policy': p, 'producer_facts': f}), parser_rejected=True)
    inc, manifest = [row['path'] for row in contract['generated_outputs']]
    changed, count = re.subn(rb'(GENI_HS_MAIN_FW_REVISION UINT32_C\()0x00000201', rb'\g<1>0x00000301', generated[inc])
    need(count == 1, 'generated revision negative operation')
    append('generated-output', 'wrong-main-revision', changed, generated[inc])
    value = document(generated[manifest]); value['scope']['physical_se6_firmware_state_known'] = True
    append('generated-output', 'false-physical-state-claim', canonical(value), generated[manifest])
    need(len(result) == 51 and sum(len(row['raw']) for row in result) == 1700682, 'complete mutant geometry')
    return result


class Archive:
    def __init__(self, path, expected_bytes, expected_sha256):
        self.path = Path(path); self.verified = {}
        digest, count = hashlib.sha256(), 0
        with self.path.open('rb') as stream:
            for block in iter(lambda: stream.read(4*1024*1024), b''):
                digest.update(block); count += len(block)
        need(count == expected_bytes and digest.hexdigest() == expected_sha256, 'full new ZIP identity')
        self.identity = {'path': str(path), 'bytes': count, 'sha256': digest.hexdigest()}
        self.archive = zipfile.ZipFile(path)
        infos = self.archive.infolist(); names = [info.filename for info in infos]
        need(len(names) == len(set(names)), 'duplicate ZIP members')
        for info in infos:
            path = PurePosixPath(info.filename)
            need(not path.is_absolute() and '..' not in path.parts and '\\' not in info.filename,
                 'unsafe ZIP member')
            need(not stat.S_ISLNK(info.external_attr >> 16), 'ZIP symlink')
        self.names = names

    def read(self, name):
        info = self.archive.getinfo(name)
        need(not info.is_dir() and info.file_size <= SMALL_LIMIT, 'bounded complete member')
        raw = self.archive.read(name)
        need(len(raw) == info.file_size, 'full member length')
        self.verified[name] = {'member': name, 'bytes': len(raw), 'sha256': sha(raw), 'git_blob_sha1': blob(raw)}
        return raw

    def binary(self, name, expected_bytes, expected_sha256):
        info = self.archive.getinfo(name)
        need(info.file_size == expected_bytes, 'QEMU member length')
        digest = hashlib.sha256(); count = 0
        with self.archive.open(name) as stream:
            for block in iter(lambda: stream.read(4*1024*1024), b''):
                if count == 0: need(block[:4] == b'\x7fELF', 'QEMU ELF magic')
                count += len(block); digest.update(block)
        need(count == expected_bytes and digest.hexdigest() == expected_sha256, 'full QEMU identity')
        self.verified[name] = {'member': name, 'bytes': count, 'sha256': digest.hexdigest()}
        return {**self.verified[name], 'executed': False}

    def case(self, name):
        matches = [path for path in self.names if re.search(r'(?:^|/)components/\d+-' + re.escape(name) + r'/case\.json$', path)]
        need(len(matches) == 1, 'unique component ' + name)
        member = matches[0]; folder = member.rsplit('/', 1)[0] + '/'
        case = document(self.read(member))
        need(case['name'] == name and case['passed'] is True and type(case['exit_code']) is int and case['exit_code'] == 0 and
             not case.get('timed_out') and 'error' not in case and 'artifact_error' not in case, 'successful component ' + name)
        logs = {label: self.read(folder + label + '.log') for label in ('stdout', 'stderr')}
        equal(case['log_sha256'], {label + '.log': sha(raw) for label, raw in logs.items()}, 'complete case log hashes')
        return member, folder, case, logs

        raw = (args.source_root / pin['path']).read_bytes(); identity(raw, pin, pin['path'])

    args = parser.parse_args()

              'auditor_sha256': sha(Path(__file__).read_bytes()), 'qemu_executed': False,
              'generator_or_verifier_imported_or_executed': False, 'guest_executed': False}
    try:
        raw = args.contract.read_bytes()

    print(json.dumps({'passed': result['passed'], 'receipt': str(args.out), 'error': result.get('error')}, sort_keys=True))
